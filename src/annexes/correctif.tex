\section*{Extrait du correctif}
Le correctif détecte l'effectif impair et attribue un exempt avant l'appariement deux à deux.

\begin{lstlisting}[language=]
// Avant : appariement suppose un effectif pair
for (let i = 0; i < equipes.length; i += 2) {
  matchs.push({ equipeA: equipes[i], equipeB: equipes[i + 1] });
}

// Apres : gestion de l'exempt si effectif impair
const aExempt = equipes.length % 2 === 1;
const exempt = aExempt ? equipes.pop() : null;
for (let i = 0; i < equipes.length; i += 2) {
  matchs.push({ equipeA: equipes[i], equipeB: equipes[i + 1] });
}
if (exempt) matchs.push({ equipeA: exempt, equipeB: null, bye: true });
\end{lstlisting}

\section*{Test de non-régression ajouté}
\begin{lstlisting}[language=]
it('attribue un exempt en ronde suisse avec un effectif impair', () => {
  const equipes = creerEquipes(5);
  const matchs = genererAppariementsRondeSuisse(equipes);
  const exempts = matchs.filter((m) => m.bye);
  expect(exempts).toHaveLength(1);
  expect(matchs).toHaveLength(3);
});
\end{lstlisting}
